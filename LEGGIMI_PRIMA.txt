<script>
const radio = document.getElementById("radio");
const play = document.getElementById("play");
const statusText = document.getElementById("status");
const songText = document.getElementById("song");
const cover = document.getElementById("cover");

async function toggleRadio() {
  if (radio.paused) {
    statusText.textContent = "Collegamento alla diretta...";
    try {
      await radio.play();
    } catch (e) {
      statusText.textContent = "Impossibile avviare la radio";
    }
  } else {
    radio.pause();
  }
}

radio.addEventListener("playing", () => {
  play.textContent = "⏸ PAUSA";
  statusText.textContent = "Stai ascoltando Radio Futura Sorso";
});

radio.addEventListener("pause", () => {
  play.textContent = "▶ PLAY";
  statusText.textContent = "Diretta in pausa";
});

radio.addEventListener("error", () => {
  statusText.textContent = "Errore di collegamento";
});

let lastSong = "";

async function updateSong() {
  try {
    const response = await fetch(
      "https://marinadisorso.radioca.st/status-json.xsl",
      {cache: "no-store"}
    );
    if (!response.ok) return;

    const data = await response.json();
    let source = data.icestats?.source;
    if (Array.isArray(source)) {
      source = source.find(s =>
        String(s.listenurl || "").includes("stream")
      ) || source[0];
    }

    const title = String(source?.title || "").trim();
    if (!title || title === lastSong) return;

    lastSong = title;
    songText.textContent = title;

    const result = await fetch(
      "https://itunes.apple.com/search?term=" +
      encodeURIComponent(title) +
      "&entity=song&limit=1"
    );
    if (!result.ok) return;

    const json = await result.json();
    const artwork = json.results?.[0]?.artworkUrl100;

    if (artwork) {
      const img = document.createElement("img");
      img.alt = "Copertina musicale";
      img.src = artwork.replace("100x100bb", "600x600bb");
      cover.replaceChildren(img);
    } else {
      cover.innerHTML =
        '<div class="placeholder">RADIO<br>FUTURA<br>SORSO</div>';
    }
  } catch (e) {
    // Se i metadati non sono accessibili,
    // la diretta continua a funzionare.
  }
}

updateSong();
setInterval(updateSong, 30000);
